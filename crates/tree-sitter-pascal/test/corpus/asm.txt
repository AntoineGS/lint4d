procedure P; assembler;
asm
        XOR     EAX,EAX
        OR      EDX,EDX
        SHR     ECX,2
        shl     eax, 2
        AND     ECX, NOT (PtrSize - 1)
        MOV     EAX,TYPE TObject
        CMP     AL,'0'
        CMP     AL,"S"
        MOV     EAX, [EAX].TFoo.Field
        jz      @Nil
@Nil:
        MOV     EAX, 10 MOD 3
        MOV     EAX, 10 DIV 3
end;
