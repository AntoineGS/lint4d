procedure Main;
begin
  if a then
    FirstLabel: SecondLabel: b
  else
    c;

  while a do
    WhileLabel: d;

  for i := a to b do
    ForLabel: e;

    WithLabel: f;

  case a of
    1: CaseLabel: g;

  EmptyLabel:
end;
