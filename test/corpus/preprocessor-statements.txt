===
Inline var inside a conditional statement block
===

procedure P;
begin
  {$IFDEF X}
  var I: Integer := 0;
  {$ENDIF}
  Foo;
end;

---

(root
  (defProc
    (declProc (kProcedure) (identifier))
    (block
      (kBegin)
      (ppBlock
        (ppIf)
        (assignment
          (varAssignDef (kVar) (identifier) (typeref (identifier)))
          (kAssign)
          (literalNumber))
        (ppEndIf))
      (statement (identifier))
      (kEnd))))
